%--------------------- Headings ---------------
% Same look as the numbered headings in the Development Plan, but with the
% Meyer book letters (G, E, S, P) in place of section numbers.
\makeatletter
\let\srs@section\section
\let\srs@subsection\subsection
\newcommand*{\book}[2]{%
    \srs@section*{#1\quad #2}%
    \addcontentsline{toc}{section}{\protect\numberline{#1}#2}}
\newcommand*{\nsection}[1]{\srs@nsplit#1 \@nil}
\def\srs@nsplit#1 #2\@nil{%
    \srs@subsection*{#1\quad #2}%
    \addcontentsline{toc}{subsection}{\protect\numberline{#1}#2}}
\let\subsection\subsubsection
\renewcommand\subsubsection{\@startsection{paragraph}{4}{\z@}%
    {3.25ex \@plus1ex \@minus.2ex}{0.5ex}{\normalfont\normalsize\bfseries}}
\makeatother
\setcounter{tocdepth}{2}
% Sub-entries (S.2.1, P.1.1, ...) are added at the subsection level like the
% \nsection entries, but without \numberline. Indent those one step further.
\makeatletter
\AtBeginDocument{%
  \let\srs@l@subsection\l@subsection
  \renewcommand*\l@subsection[2]{%
    \@expandtwoargs\in@{\detokenize{\numberline}}{\detokenize{#1}}%
    \ifin@ \srs@l@subsection{#1}{#2}%
    \else \@dottedtocline{2}{3.8em}{3.2em}{#1}{#2}\fi}}
\makeatother
